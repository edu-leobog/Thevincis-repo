import random

tal = random.randint(1, 100)

gissning = int(input("Gissa ett tal mellan 1 och 100: "))

while gissning != tal:
    if gissning < tal:
        print("För lågt! Försök igen.")
    else:
        print("För högt! Försök igen.")
    
    gissning = int(input("Gissa ett tal mellan 1 och 100: "))   

print("Grattis! Du gissade rätt!")
